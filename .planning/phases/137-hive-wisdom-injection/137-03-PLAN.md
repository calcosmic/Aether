       ```typescript
       async function prepareHiveSection(bridge: GoBridgeOptions): Promise<string> {
         try {
           const domains = await resolveDomainTags(bridge);
           return await readHiveWisdom({
             goBinaryPath: bridge.goBinaryPath,
             cwd: bridge.cwd,
             domains,
             minConfidence: 0.5,
             maxEntries: 10,
             budgetChars: 2500,
           });
         } catch (err: unknown) {
           const msg = err instanceof Error ? err.message : String(err);
           process.stderr.write(`Warning: hive-read failed: ${msg}\n`);
           return "";
         }
       }
       ```
       This helper encapsulates the D-04/D-05 graceful degradation: any failure in resolveDomainTags or readHiveWisdom logs a warning and returns empty string.
