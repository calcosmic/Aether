2. If any command lacks a dedicated test, add a lightweight smoke test

Smoke test pattern:
```go
func TestBuild_Smoke(t *testing.T) {
    saveGlobals(t)
    resetRootCmd(t)
    s, tmpDir := newTestStore(t)
    defer os.RemoveAll(tmpDir)
    store = s

    rootCmd.SetArgs([]string{"build", "--help"})
    err := rootCmd.Execute()
    if err != nil {
        t.Fatalf("build --help returned error: %v", err)
    }
    // Verify help output contains expected text
}
```

Or for commands that need state:
```go
func TestStatus_Smoke(t *testing.T) {
    saveGlobals(t)
    resetRootCmd(t)
    s, tmpDir := newTestStore(t)
    defer os.RemoveAll(tmpDir)
    store = s

    // Initialize minimal colony state
    // ... setup code ...

    rootCmd.SetArgs([]string{"status", "--json"})
    err := rootCmd.Execute()
    if err != nil {
        t.Fatalf("status --json returned error: %v", err)
    }
    // Verify JSON output
}
```
</action>

<acceptance_criteria>

- Tests pass: go test ./cmd/... -run 'TestBuild|TestContinue|TestPlan|TestSeal|TestEntomb|TestColonize|TestStatus|TestOracle|TestSwarm' -count=1
</acceptance_criteria>

Generate the matrix programmatically:
```bash

c = json.load(open('cmd/testdata/command_catalog.json'))
for e in c:
```
